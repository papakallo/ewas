\subsection{Terms, Definitions and Abbreviations}

\subsubsection{Project-Specific Terms and Definitions}

\begin{itemize}[leftmargin=1.5em, itemsep=0.25em, topsep=0.2em]
    \item \textbf{As-Designed Baseline}: The technical configuration of Configuration Item \texttt{CI-03} established during Phase~A/B to satisfy the functional, performance, interface, and budget requirements of \textbf{AD-01}, prior to physical qualification and as-built verification.
    \item \textbf{Careful COTS}: A parts selection and assurance philosophy for the 72-satellite \abbr{SATL} \abbr{LEO} layer that combines commercial/automotive-grade components with targeted radiation lot screening, localized \(\mathrm{Al}/\mathrm{Ta}\) spot shielding, \abbr{EDAC}/\abbr{TMR} protection, and hardware watchdog \abbr{SEL} current-limiting to balance 7-year reliability against constellation unit cost.
    \item \textbf{Proton Despiking}: Real-time on-board temporal and spatial filtering applied within the \abbr{SATL}/\abbr{SATG} infrared focal plane read-out pipeline to reject transient high-energy proton charge-deposition spikes without triggering false boost-phase plume tracks.
    \item \textbf{South Atlantic Anomaly (SAA)}: A geomagnetic dip region over the South Atlantic and South America where the inner Van Allen radiation belt approaches low Earth orbit (\(\sim 200\text{--}1000\text{ km}\)), exposing \abbr{SATL} platforms at \(1{,}000\text{ km}\) circular altitude to elevated trapped proton fluxes requiring heightened \abbr{SEU}/\abbr{SEL} mitigation and focal-plane despiking.
\end{itemize}
