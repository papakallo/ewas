\section{Annex B: Verification Control and Compliance Matrix}

This annex defines the formal verification methodology, verification levels, model philosophy, milestone review stages, and Master Verification Control Matrix (Table~\ref{tab:annex_b_vcd}) governing all 77 normative technical requirements of Configuration Item \textbf{\texttt{CI-03}: Space Segment (\abbr{SSEG}, \abbr{WBS} Element \texttt{123456.03})} (\textbf{ECSS-E-ST-10-02C}, \textbf{AD-04}; \textbf{ECSS-E-ST-10-03C}, \textbf{AD-05}; \textbf{ECSS-E-HB-10-02A}, \textbf{RD-06}).

\subsection{B.1 Space Segment Verification Methodology, Levels, Models, and Stages}

\begin{enumerate}
    \item \textbf{ECSS Verification Methods (\textbf{ECSS-E-ST-10-02C} Clause 5.2):}
    \begin{itemize}
        \item \textbf{\texttt{T} --- Test:} Verification by quantitative measurement of functional, performance, electro-optical, radio-frequency/optical link, or environmental parameters under controlled excitation using hardware/software models (\texttt{EM}, \texttt{STM}, \texttt{QM}/\texttt{PFM}, \texttt{FM}) or in-orbit commissioning trials. Documented in a formal \textbf{Test Report (\texttt{TRPT})}.
        \item \textbf{\texttt{A} --- Analysis:} Verification by theoretical, astrodynamic, radiometric, finite-element structural/thermal, radiation transport, or reliability simulation using validated numerical models (\texttt{GMAT}, \texttt{VTS}, \texttt{ESATAN}, \texttt{NASTRAN}, \texttt{FASTRAD}, \texttt{MASTER-8}). Documented in a formal \textbf{Analysis Report (\texttt{ARPT})}.
        \item \textbf{\texttt{R} --- Review of Design:} Verification by validation of approved design drawings, system architectures, flight software source code/schemas, cryptographic protocols, and Design Definition/Justification Files (\texttt{DDF}/\texttt{DDJF}). Documented in a formal \textbf{Review-of-Design Report (\texttt{RRPT})}.
        \item \textbf{\texttt{I} --- Inspection:} Verification by physical, dimensional, visual, or interface gauge examination of manufactured hardware, material certificates, safety inhibits, and connector/bolt layouts against approved \abbr{ICD} drawings. Documented in a formal \textbf{Inspection Report (\texttt{IRPT})}.
    \end{itemize}
    \item \textbf{Verification Levels and Model Philosophy (\textbf{ECSS-E-ST-10-03C}, \abbr{WBS} \texttt{123456.04.02}):}
    \begin{itemize}
        \item \textbf{\texttt{L2-SSEG} (Space Segment / Constellation Level):} Verified across the integrated 6~\abbr{GEO} + 72~\abbr{LEO} architecture using the End-to-End Mission Simulator (\texttt{SIM}), Avionics/ISL Hardware-in-the-Loop testbed (\texttt{HIL}), and In-Orbit Commissioning campaign (\texttt{IOC}).
        \item \textbf{\texttt{L3-SC} (Spacecraft Level --- \texttt{SATL} \& \texttt{SATG}):} Verified on the Structural-Thermal Model (\texttt{STM}), lead Qualification / Proto-Flight Models (\texttt{QM}/\texttt{PFM}), and recurring Flight Models (\texttt{FM-01}--\texttt{FM-72} for \texttt{SATL}; \texttt{FM-01}--\texttt{FM-06} for \texttt{SATG}).
        \item \textbf{\texttt{L4-SUB} (Subsystem / Equipment Level):} Verified at unit level on Engineering Models (\texttt{EM}) and Qualification/Flight units (\abbr{PLD} optical bench, \abbr{ADCS} star trackers/\abbr{FSM}, \abbr{OBDH} boards, and 1,550~nm \abbr{OISL} terminals).
    \end{itemize}
    \item \textbf{ECSS Verification Stages (\textbf{ECSS-E-ST-10-02C} Clause 5.4):}
    \textbf{\texttt{PDR}} (\emph{Preliminary Design Review}), \textbf{\texttt{CDR}} (\emph{Critical Design Review}), \textbf{\texttt{QR}} (\emph{Qualification Review}), \textbf{\texttt{AR}} (\emph{Acceptance Review} prior to launch shipment), and \textbf{\texttt{CR/ORR}} (\emph{In-Orbit Commissioning \& Operational Readiness Review}).
\end{enumerate}
